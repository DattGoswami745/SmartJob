CREATE TABLE Users (
    UserId INT IDENTITY(1,1) PRIMARY KEY,
    FullName NVARCHAR(100) NOT NULL,
    Email NVARCHAR(150) NOT NULL UNIQUE,
    PasswordHash NVARCHAR(255) NOT NULL,
    Role NVARCHAR(50) DEFAULT 'User',
    CreatedAt DATETIME DEFAULT GETDATE(),
    IsActive BIT DEFAULT 1,
    IsEmailVerified BIT DEFAULT 0 NOT NULL,
    EmailOTP VARCHAR(255),
    EmailOTPExpiry DATETIME,
    CompanyId INT
);

CREATE TABLE UserProfiles (
    ProfileId INT IDENTITY(1,1) PRIMARY KEY,
    UserId INT NOT NULL,
    Skills NVARCHAR(MAX),
    ExperienceYears INT,
    Education NVARCHAR(150),
    PreferredLocation NVARCHAR(100),
    ResumePath NVARCHAR(255),
    ResumeFile VARBINARY(MAX),
    ResumeFileName NVARCHAR(255),
    ResumeContentType NVARCHAR(100),

    CONSTRAINT FK_UserProfiles_Users 
    FOREIGN KEY (UserId) REFERENCES Users(UserId)
);

CREATE TABLE UserActivityLogs (
    LogId INT IDENTITY(1,1) PRIMARY KEY,
    UserId INT,
    Action NVARCHAR(100),
    ActionDate DATETIME DEFAULT GETDATE(),

    CONSTRAINT FK_UserActivityLogs_Users
    FOREIGN KEY (UserId) REFERENCES Users(UserId)
);

CREATE TABLE ReportGenerationLogs (
    LogId INT IDENTITY(1,1) PRIMARY KEY,
    ReportId INT,
    UserId INT,
    GeneratedAt DATETIME DEFAULT GETUTCDATE(),
    Format NVARCHAR(20),
    FilterValues NVARCHAR(MAX),

    CONSTRAINT FK_ReportGenerationLogs_Report
    FOREIGN KEY (ReportId) REFERENCES ReportConfigurations(ReportId)
);

CREATE TABLE ReportConfigurations (
    ReportId INT IDENTITY(1,1) PRIMARY KEY,
    ReportName NVARCHAR(200) NOT NULL,
    Description NVARCHAR(MAX),
    BaseTable NVARCHAR(MAX) NOT NULL,
    SelectedFields NVARCHAR(MAX) NOT NULL,
    Filters NVARCHAR(MAX),
    IsActive BIT DEFAULT 1,
    CreatedBy INT,
    CreatedAt DATETIME DEFAULT GETUTCDATE(),
    UpdatedAt DATETIME DEFAULT GETUTCDATE()
);

CREATE TABLE Parameters (
    ParamKey NVARCHAR(100) PRIMARY KEY,
    ParamValue NVARCHAR(MAX) NOT NULL,
    Description NVARCHAR(255)
);

CREATE TABLE Jobs (
    JobId INT IDENTITY(1,1) PRIMARY KEY,
    CompanyId INT NOT NULL,
    Title NVARCHAR(150) NOT NULL,
    Description NVARCHAR(MAX) NOT NULL,
    RequiredSkills NVARCHAR(MAX) NOT NULL,
    JobType NVARCHAR(50),
    SalaryRange NVARCHAR(50),
    PostedDate DATETIME DEFAULT GETDATE(),
    IsActive BIT DEFAULT 1,
    LastDate DATETIME,
    IsApproved BIT DEFAULT 0 NOT NULL,
    JobDescriptionFile NVARCHAR(MAX),
    JobDescriptionText NVARCHAR(MAX),
    JobDescriptionUpdatedAt DATETIME,

    CONSTRAINT FK_Jobs_Companies
    FOREIGN KEY (CompanyId) REFERENCES Companies(CompanyId)
);

CREATE TABLE CompanyVerificationDocuments (
    DocumentId BIGINT IDENTITY(1,1) PRIMARY KEY,
    CompanyId BIGINT NOT NULL,
    DocumentType VARCHAR(100) NOT NULL,
    FileName VARCHAR(500) NOT NULL,
    FilePath VARCHAR(1000) NOT NULL,
    IsVerified BIT DEFAULT 0,
    VerifiedBy BIGINT,
    VerifiedOnUTC DATETIME,
    IsRejected BIT DEFAULT 0,
    RejectReason VARCHAR(500),
    RecordedBy BIGINT,
    RecordedOnUTC DATETIME DEFAULT GETUTCDATE(),
    DocumentFile VARBINARY(MAX),
    DocumentContentType NVARCHAR(100)
);

CREATE TABLE Companies (
    CompanyId INT IDENTITY(1,1) PRIMARY KEY,
    CompanyName NVARCHAR(150) NOT NULL,
    Industry NVARCHAR(100),
    Location NVARCHAR(100),
    CreatedAt DATETIME DEFAULT GETDATE(),
    IsCompanyVerified BIT DEFAULT 0
);

CREATE TABLE Applications (
    ApplicationId INT IDENTITY(1,1) PRIMARY KEY,
    JobId INT NOT NULL,
    UserId INT NOT NULL,
    ApplicationStatus NVARCHAR(50) DEFAULT 'Applied',
    AppliedDate DATETIME DEFAULT GETDATE(),

    CONSTRAINT FK_Applications_Jobs
    FOREIGN KEY (JobId) REFERENCES Jobs(JobId),

    CONSTRAINT FK_Applications_Users
    FOREIGN KEY (UserId) REFERENCES Users(UserId)
);
